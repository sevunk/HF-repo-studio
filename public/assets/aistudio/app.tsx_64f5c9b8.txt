function formatSize(bytes?: number): string {
  if (bytes === undefined || bytes === null || isNaN(bytes)) return '0 B';
  if (bytes >= 1024 ** 3) {
    return `${(bytes / 1024 ** 3).toFixed(2)} GiB (${(bytes / 1e9).toFixed(2)} GB)`;
  }
  if (bytes >= 1024 ** 2) {
    return `${(bytes / 1024 ** 2).toFixed(2)} MB`;
  }
  if (bytes >= 1024) {
    return `${(bytes / 1024).toFixed(1)} KB`;
  }
  return `${bytes} B`;
}

function formatWib(isoDate?: string): string {
  if (!isoDate) return '-';
  try {
    const dt = new Date(isoDate);
    return new Intl.DateTimeFormat('id-ID', {
      timeZone: 'Asia/Jakarta',
      year: 'numeric',
      month: '2-digit',
      day: '2-digit',
      hour: '2-digit',
      minute: '2-digit',
      second: '2-digit',
      hour12: false,
    }).format(dt) + ' WIB';
  } catch {
    return isoDate;
  }
}

function parseHfInput(
  rawRepoOrUrl: string,
  rawSubpath: string,
  defaultType: RepoType,
  defaultRev: string
): { repoId: string; repoType: RepoType; revision: string; subpath: string } {
  const raw = (rawRepoOrUrl || '').trim();
  const sub = (rawSubpath || '').trim().replace(/^\/+|\/+$/g, '');
  let repoType: RepoType = defaultType;
  let revision = (defaultRev || 'main').trim() || 'main';

  if (raw.startsWith('http://') || raw.startsWith('https://')) {
    const cleanUrl = raw.split('?')[0].replace(/\/+$/, '');
    const parsed = new URL(cleanUrl);
    let parts = parsed.pathname
      .split('/')
      .filter(Boolean)
      .map((p) => decodeURIComponent(p));

    if (parts.length > 0 && ['datasets', 'spaces', 'models'].includes(parts[0])) {
      const map: Record<string, RepoType> = {
        datasets: 'dataset',
        spaces: 'space',
        models: 'model',
      };
      repoType = map[parts[0]];
      parts = parts.slice(1);
    }

    if (parts.length < 2) {
      throw new Error(`URL Hugging Face tidak valid: ${raw}`);
    }

    const repoId = `${parts[0]}/${parts[1]}`;
    let urlSubpath = '';
    if (parts.length >= 4 && ['blob', 'resolve', 'tree', 'commits'].includes(parts[2])) {
      revision = parts[3];
      urlSubpath = parts.slice(4).join('/');
    } else if (parts.length > 2) {
      urlSubpath = parts.slice(2).join('/');
    }

    return {
      repoId,
      repoType,
      revision,
      subpath: sub || urlSubpath,
    };
  }

  const cleanRaw = raw.startsWith('hf://') ? raw.slice(5) : raw;
  let parts = cleanRaw.split('/').filter(Boolean);
  if (parts.length > 0 && ['datasets', 'spaces', 'models'].includes(parts[0])) {
    const map: Record<string, RepoType> = {
      datasets: 'dataset',
      spaces: 'space',
      models: 'model',
    };
    repoType = map[parts[0]];
    parts = parts.slice(1);
  }

  if (parts.length < 2) {
    throw new Error(`Format repo_id tidak valid: "${raw}". Gunakan format "username/nama-repo".`);
  }

  const repoId = `${parts[0]}/${parts[1]}`;
  const inlineSubpath = parts.length > 2 ? parts.slice(2).join('/') : '';
  return {
    repoId,
    repoType,
    revision,
    subpath: sub || inlineSubpath,
  };
}

function arrayBufferToBase64(buffer: ArrayBuffer): string {
  const bytes = new Uint8Array(buffer);
  let binary = '';
  const chunkSize = 0x8000;
  for (let i = 0; i < bytes.length; i += chunkSize) {
    binary += String.fromCharCode(...bytes.subarray(i, i + chunkSize));
  }
  return btoa(binary);
}

function repoApiPrefix(repoType: RepoType, repoId: string): string {
  const [owner, name] = repoId.split('/');
  return `${HF_BASE}/api/${repoType}s/${encodeURIComponent(owner)}/${encodeURIComponent(name)}`;
}

function repoWebUrl(repoType: RepoType, repoId: string, rev = 'main', subpath = ''): string {
  const prefix = repoType === 'dataset' ? 'datasets/' : repoType === 'space' ? 'spaces/' : '';
  const cleanSub = subpath ? `/${subpath.replace(/^\/+/, '')}` : '';
  return `${HF_BASE}/${prefix}${repoId}/tree/${encodeURIComponent(rev)}${cleanSub}`;
}

function repoResolveUrl(repoType: RepoType, repoId: string, rev: string, filePath: string): string {
  const prefix = repoType === 'dataset' ? 'datasets/' : repoType === 'space' ? 'spaces/' : '';
  const encodedPath = filePath.split('/').map(encodeURIComponent).join('/');
  return `${HF_BASE}/${prefix}${repoId}/resolve/${encodeURIComponent(rev)}/${encodedPath}`;
}

async function hfFetchJson(url: string, token: string, init?: RequestInit) {
  const headers: Record<string, string> = {
    Accept: 'application/json',
    ...(init?.headers as Record<string, string>),
  };
  if (token.trim()) {
    headers['Authorization'] = `Bearer ${token.trim()}`;
  }
  const resp = await fetch(url, { ...init, headers });
  if (!resp.ok) {
    let errText = '';
    try {
      const j = await resp.json();
      errText = j.error || JSON.stringify(j);
    } catch {
      errText = await resp.text();
    }
    throw new Error(`HTTP ${resp.status}: ${errText || resp.statusText}`);
  }
  const text = await resp.text();
  return text ? JSON.parse(text) : {};
}

async function fetchCommitFileMap(
  repoId: string,
  repoType: RepoType,
  revision: string,
  token: string
): Promise<Record<string, HfFileItem>> {
  const url = `${repoApiPrefix(repoType, repoId)}/tree/${encodeURIComponent(revision)}?recursive=true`;
  const tree = await hfFetchJson(url, token);
  const map: Record<string, HfFileItem> = {};
  if (!Array.isArray(tree)) return map;

  for (const item of tree) {
    if (item.type === 'file' && item.path !== '.gitattributes') {
      const lfs = item.lfs;
      const isLfs = Boolean(lfs && lfs.oid);
      const fullHash = isLfs ? String(lfs.oid) : String(item.oid || '-');
      map[item.path] = {
        path: item.path,
        size: Number(lfs?.size ?? item.size ?? 0),
        hash: fullHash.slice(0, 10),
        fullHash,
        isLfs,
        lfsOid: lfs?.oid,
        lfsSize: lfs?.size ?? item.size,
        xetHash: item.xetHash,
      };
    }
  }
  return map;
}

async function fetchPathsInfoDetailed(
  repoId: string,
  repoType: RepoType,
  revision: string,
  paths: string[],
  token: string
): Promise<Record<string, HfFileItem>> {
  const result: Record<string, HfFileItem> = {};
  if (paths.length === 0) return result;

  const url = `${repoApiPrefix(repoType, repoId)}/paths-info/${encodeURIComponent(revision)}`;
  const data = await hfFetchJson(url, token, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ paths, expand: true }),
  });

  if (Array.isArray(data)) {
    for (const item of data) {
      if (item && item.path) {
        const lfs = item.lfs;
        const isLfs = Boolean(lfs && lfs.oid);
        const fullHash = isLfs ? String(lfs.oid) : String(item.oid || '-');
        result[item.path] = {
          path: item.path,
          size: Number(lfs?.size ?? item.size ?? 0),
          hash: fullHash.slice(0, 10),
          fullHash,
          isLfs,
          lfsOid: lfs?.oid,
          lfsSize: lfs?.size ?? item.size,
          xetHash: item.xetHash,
        };
      }
    }
  }
  return result;
}

async function ensureRepoExists(
  repoId: string,
  repoType: RepoType,
  isPrivate: boolean,
  token: string
) {
  const [owner, name] = repoId.split('/');
  if (!owner || !name) throw new Error(`Format repo tujuan tidak valid: ${repoId}`);

  // Cek apakah sudah ada
  const checkUrl = `${repoApiPrefix(repoType, repoId)}`;
  const headers: Record<string, string> = { Accept: 'application/json' };
  if (token.trim()) headers['Authorization'] = `Bearer ${token.trim()}`;
  const checkResp = await fetch(checkUrl, { headers });
  if (checkResp.ok) return;

  // Buat repo baru jika belum ada
  const whoami = await hfFetchJson(`${HF_BASE}/api/whoami-v2`, token);
  const currentUser = whoami.name || '';
  const bodyPayload: Record<string, any> = {
    name,
    type: repoType,
    private: isPrivate,
  };
  if (owner.toLowerCase() !== currentUser.toLowerCase()) {
    bodyPayload.organization = owner;
  }

  const createResp = await fetch(`${HF_BASE}/api/repos/create`, {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${token.trim()}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(bodyPayload),
  });

  if (!createResp.ok && createResp.status !== 409) {
    const errTxt = await createResp.text();
    throw new Error(`Gagal membuat repositori ${repoId}: HTTP ${createResp.status} ${errTxt}`);
  }
}

async function executeNdjsonCommit(opts: {
  srcRepoId: string;
  srcRepoType: RepoType;
  srcRevision: string;
  dstRepoId: string;
  dstRepoType: RepoType;
  dstRevision: string;
  copyPairs: { srcFile: HfFileItem; dstPath: string }[];
  deletePaths: { path: string; isFolder: boolean }[];
  commitMessage: string;
  commitDescription?: string;
  token: string;
  onLog?: (msg: string) => void;
}) {
  const {
    srcRepoId, srcRepoType, srcRevision,
    dstRepoId, dstRepoType, dstRevision,
    copyPairs, deletePaths, commitMessage, commitDescription, token, onLog
  } = opts;

  const sameRepo =
    srcRepoId.toLowerCase() === dstRepoId.toLowerCase() && srcRepoType === dstRepoType;

  // 1. Jika lintas repo dan ada file LFS, panggil endpoint lfs-files/duplicate untuk Server-Side Copy 0 GB
  const lfsPairs = copyPairs.filter((p) => p.srcFile.isLfs && p.srcFile.lfsOid);
  if (!sameRepo && lfsPairs.length > 0) {
    onLog?.(`⚡ Mengonfigurasi Server-Side LFS/Xet Duplicate untuk ${lfsPairs.length} file besar...`);
    const detailedMap = await fetchPathsInfoDetailed(
      srcRepoId,
      srcRepoType,
      srcRevision,
      lfsPairs.map((p) => p.srcFile.path),
      token
    );

    const filesToDuplicate = lfsPairs
      .map((p) => {
        const d = detailedMap[p.srcFile.path] || p.srcFile;
        return {
          xetHash: d.xetHash,
          sha256: d.lfsOid || d.fullHash,
          filename: p.srcFile.path,
        };
      })
      .filter((x) => x.sha256);

    if (filesToDuplicate.length > 0) {
      const dupUrl = `${repoApiPrefix(srcRepoType, srcRepoId)}/lfs-files/duplicate`;
      try {
        await hfFetchJson(dupUrl, token, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            target: { type: dstRepoType, name: dstRepoId },
            files: filesToDuplicate,
          }),
        });
      } catch (dupErr: any) {
        onLog?.(`ℹ️ Info storage duplicate: ${dupErr.message} (Melanjutkan commit pointer LFS...)`);
      }
    }
  }

  // 2. Susun baris NDJSON
  const records: any[] = [
    {
      key: 'header',
      value: {
        summary: commitMessage,
        description: commitDescription || '',
      },
    },
  ];

  for (const pair of copyPairs) {
    const f = pair.srcFile;
    if (f.isLfs && f.lfsOid) {
      records.push({
        key: 'lfsFile',
        value: {
          path: pair.dstPath,
          algo: 'sha256',
          oid: f.lfsOid,
          size: f.lfsSize ?? f.size,
        },
      });
    } else {
      // File non-LFS kecil (misal README.md, .json, .csv): ambil via resolve dan encode base64
      onLog?.(`📄 Menyalin file teks/non-LFS (${formatSize(f.size)}): ${f.path} ...`);
      const rawUrl = repoResolveUrl(srcRepoType, srcRepoId, srcRevision, f.path);
      const headers: Record<string, string> = {};
      if (token.trim()) headers['Authorization'] = `Bearer ${token.trim()}`;
      const resp = await fetch(rawUrl, { headers });
      if (!resp.ok) {
        throw new Error(`Gagal membaca file non-LFS ${f.path} (HTTP ${resp.status})`);
      }
      const buf = await resp.arrayBuffer();
      const b64 = arrayBufferToBase64(buf);
      records.push({
        key: 'file',
        value: {
          content: b64,
          path: pair.dstPath,
          encoding: 'base64',
        },
      });
    }
  }

  for (const del of deletePaths) {
    records.push({
      key: del.isFolder ? 'deletedFolder' : 'deletedFile',
      value: { path: del.path },
    });
  }

  const ndjsonBody = records.map((r) => JSON.stringify(r)).join('\n') + '\n';
  const commitUrl = `${repoApiPrefix(dstRepoType, dstRepoId)}/commit/${encodeURIComponent(dstRevision)}`;

  onLog?.(`🚀 Mengirim Atomic Commit ke ${dstRepoId} (${dstRevision})...`);
  return await hfFetchJson(commitUrl, token, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-ndjson' },
    body: ndjsonBody,
  });
}

          const deleted: NonNullable<HfCommitInfo['diff']>['deleted'] = [];
          const overwritten: NonNullable<HfCommitInfo['diff']>['overwritten'] = [];

          for (const [p, fInfo] of Object.entries(currFiles)) {
            if (!(p in prevFiles)) {
              added.push(fInfo);
            } else if (fInfo.fullHash !== prevFiles[p].fullHash) {
              overwritten.push({
                path: p,
                oldFile: prevFiles[p],
                newFile: fInfo,
                rescueCommitId: prevCommit.id,
                rescueIndex: idx + 1,
              });
              hints.push({
                type: 'OVERWRITTEN',
                path: p,
                happenedAtCommitIdx: idx,
                happenedAtShortId: commitObj.shortId,
                rescueCommitIdx: idx + 1,
                rescueCommitId: prevCommit.id,
                rescueShortId: prevShortId,
                oldFile: prevFiles[p],
                newFile: fInfo,
              });
            }
          }

          for (const [p, oldInfo] of Object.entries(prevFiles)) {
            if (!(p in currFiles)) {
              deleted.push({
                file: oldInfo,
                rescueCommitId: prevCommit.id,
                rescueIndex: idx + 1,
              });
              hints.push({
                type: 'DELETED',
                path: p,
                happenedAtCommitIdx: idx,
                happenedAtShortId: commitObj.shortId,
                rescueCommitIdx: idx + 1,
                rescueCommitId: prevCommit.id,
                rescueShortId: prevShortId,
                oldFile: oldInfo,
              });
            }
          }

          commitObj.diff = { added, deleted, overwritten };
        }

        builtCommits.push(commitObj);
      }

      setExpandedCommits(defaultExpanded);
      setTmCommits(builtCommits);
      setTmRescueHints(hints);
    } catch (err: any) {
      setTmError(err.message || String(err));

                                    <td className="py-2 px-2 text-right whitespace-nowrap">
                                      <button
                                        onClick={() => handleSelectForRestore(c.commitId, f.path)}

                            </table>

                      </div>

                >
                  <h3 className="font-bold text-sm flex items-center gap-2 text-amber-400">
                    <RotateCcw className="w-4 h-4" />
                    2. Panel Eksekusi Pemulihan (Restore)
                  </h3>
                  <p className="text-[11px] opacity-75">
                    Klik tombol <strong>"🚑 Pulihkan Ini"</strong> pada daftar file di kiri, atau isi manual di bawah ini. Proses berjalan 100% Server-Side (0 GB Download).
                  </p>

                    <label className="block text-xs font-medium mb-1">
                      Commit ID atau Nomor Urut <code>[index]</code>
                    </label>
                    <input
                      type="text"
                      value={restoreCommitId}
                      onChange={(e) => setRestoreCommitId(e.target.value)}

                      className={`w-full px-3 py-2 rounded-xl border text-xs font-mono ${
                        isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'

                    />
                  </div>

                    <label className="block text-xs font-medium mb-1">

                      className={`w-full px-3 py-2 rounded-xl border text-xs font-mono ${

                    <label className="block text-xs font-medium mb-1">
                      Jika Nama File Sudah Ada di Repo Saat Ini:
                    </label>
                    <select
                      value={restoreConflictMode}
                      onChange={(e) => setRestoreConflictMode(e.target.value as any)}
                      className={`w-full px-3 py-2 rounded-xl border text-xs ${
                        isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                      }`}
                    >
                      <option value="OVERWRITE">TIMPA & Kembalikan Versi Commit Lama</option>
                      <option value="SUFFIX_PULIH">Jangan Timpa (Simpan dengan Akhiran _PULIH)</option>
                      <option value="ABORT">Batalkan Jika Nama Sudah Ada</option>
                    </select>
                  </div>

                  <div>
                    <label className="block text-xs font-medium mb-1">
                      Sekalian Hapus File Lain / Salah (Opsional, pisah koma)
                    </label>
                    <input
                      type="text"
                      value={restoreExtraDeletes}
                      onChange={(e) => setRestoreExtraDeletes(e.target.value)}
                      placeholder="file_salah.safetensors"
                      className={`w-full px-3 py-2 rounded-xl border text-xs font-mono ${
                        isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                      }`}
                    />
                  </div>

                  <button
                    onClick={handleExecuteRestore}
                    disabled={restoreStatus.loading}
                    className="w-full py-2.5 px-4 rounded-xl bg-emerald-600 hover:bg-emerald-500 disabled:opacity-50 text-white font-bold text-xs shadow-lg shadow-emerald-600/25 flex items-center justify-center gap-2 transition"
                  >
                    <RotateCcw className="w-4 h-4" />
                    {restoreStatus.loading ? 'Memulihkan di Server HF...' : '🚑 Eksekusi Pulihkan Sekarang'}
                  </button>

                  {/* Log Status Restore */}
                  {(restoreStatus.logs.length > 0 || restoreStatus.error) && (
                    <div className={`p-3 rounded-xl border text-[11px] font-mono space-y-1.5 ${
                      isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-200'
                    }`}>
                      {restoreStatus.logs.map((l, i) => (
                        <div key={i} className="leading-relaxed">{l}</div>
                      ))}
                      {restoreStatus.error && (
                        <div className="text-red-400 font-bold pt-1">❌ {restoreStatus.error}</div>
                      )}
                      {restoreStatus.successUrl && (
                        <a
                          href={restoreStatus.successUrl}
                          target="_blank"
                          rel="noreferrer"
                          className="mt-2 inline-flex items-center gap-1.5 text-emerald-400 font-sans font-bold hover:underline"
                        >
                          🔗 Buka Repo Hasil Pemulihan <ExternalLink className="w-3.5 h-3.5" />
                        </a>
                      )}
                    </div>
                  )}
                </div>
              </div>
            )}
          </div>
        )}

        {/* =================================================================== */}
        {/* 🚀 TAB 2: HF REPO-TO-REPO MOVER, COPIER & RENAMER                   */}
        {/* =================================================================== */}
        {activeStudioTab === 'mover' && (
          <div className="space-y-6">
            <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
              {/* PANEL KIRI: SUMBER (REPO ASAL) */}
              <div className={`rounded-2xl border p-5 shadow-sm space-y-4 ${
                isDark ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'
              }`}>
                <div className="flex items-center justify-between flex-wrap gap-2">
                  <h2 className="font-bold text-sm sm:text-base flex items-center gap-2">
                    <FileBox className="w-5 h-5 text-indigo-400" />
                    1. Sumber File / Folder (Repo Asal)
                  </h2>
                  {userRepos.length > 0 && (
                    <select
                      onChange={(e) => {
                        if (!e.target.value) return;
                        const found = userRepos.find((r) => r.id === e.target.value);
                        if (found) {
                          setMvSrcRepoInput(found.id);
                          setMvSrcType(found.type);
                        }
                      }}
                      value=""
                      className={`px-3 py-1 rounded-xl border text-xs ${
                        isDark ? 'bg-slate-950 border-slate-800 text-indigo-400' : 'bg-slate-50 border-slate-300'
                      }`}
                    >
                      <option value="">⚡ Pilih Repo Asal Saya...</option>
                      {userRepos.map((r) => (
                        <option key={`src-${r.type}-${r.id}`} value={r.id}>
                          [{r.type.toUpperCase()}] {r.id}
                        </option>
                      ))}
                    </select>
                  )}
                </div>

                <div className="space-y-3">
                  <div>
                    <label className="block text-xs font-medium mb-1 opacity-80">
                      Link URL File/Repo HF <strong>ATAU</strong> Nama Repo Asal (`username/repo`)
                    </label>
                    <input
                      type="text"
                      list="user-hf-repos"
                      value={mvSrcRepoInput}
                      onChange={(e) => setMvSrcRepoInput(e.target.value)}
                      placeholder="https://huggingface.co/SEVUNX/redkrea/resolve/main/model.safetensors"
                      className={`w-full px-3.5 py-2 rounded-xl border text-xs font-mono ${
                        isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                      }`}
                    />
                  </div>

                  <div className="grid grid-cols-2 gap-3">
                    <div>
                      <label className="block text-xs font-medium mb-1 opacity-80">Tipe Repo Asal</label>
                      <select
                        value={mvSrcType}
                        onChange={(e) => setMvSrcType(e.target.value as RepoType)}
                        className={`w-full px-3 py-2 rounded-xl border text-xs ${
                          isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                        }`}
                      >
                        <option value="model">model</option>
                        <option value="dataset">dataset</option>
                        <option value="space">space</option>
                      </select>
                    </div>
                    <div>
                      <label className="block text-xs font-medium mb-1 opacity-80">Branch Asal</label>
                      <input
                        type="text"
                        value={mvSrcBranch}
                        onChange={(e) => setMvSrcBranch(e.target.value)}
                        className={`w-full px-3 py-2 rounded-xl border text-xs font-mono ${
                          isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                        }`}
                      />
                    </div>
                  </div>

                  <div>
                    <label className="block text-xs font-medium mb-1 opacity-80">
                      Path File / Folder di Repo Asal (Opsional jika sudah pakai URL atau centang tabel di bawah)
                    </label>
                    <div className="flex gap-2">
                      <input
                        type="text"
                        value={mvSrcPathInput}
                        onChange={(e) => {
                          setMvSrcPathInput(e.target.value);
                          setMvSelectedPaths(new Set());
                        }}
                        placeholder="Contoh: model.safetensors atau ketik * untuk semua file"
                        className={`flex-1 px-3.5 py-2 rounded-xl border text-xs font-mono ${
                          isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                        }`}
                      />
                      <button
                        onClick={handleLoadSourceFiles}
                        disabled={mvLoadingFiles}
                        className="px-3.5 py-2 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white text-xs font-bold flex items-center gap-1.5 shrink-0 transition"
                      >
                        <RefreshCw className={`w-3.5 h-3.5 ${mvLoadingFiles ? 'animate-spin' : ''}`} />
                        Muat Daftar File
                      </button>
                    </div>
                  </div>
                </div>

                {/* Pemilih File Visual Repo Asal */}
                {Object.keys(mvSrcFilesMap).length > 0 && (
                  <div className={`p-3.5 rounded-xl border space-y-2.5 ${
                    isDark ? 'bg-slate-950/70 border-slate-800' : 'bg-slate-50 border-slate-200'
                  }`}>
                    <div className="flex items-center justify-between gap-2">
                      <span className="text-xs font-bold text-indigo-400">
                        📂 Pilih / Centang File dari Repo Asal ({mvSelectedPaths.size} dipilih)
                      </span>
                      <div className="flex items-center gap-2">
                        <button
                          onClick={() => setMvSelectedPaths(new Set(Object.keys(mvSrcFilesMap)))}
                          className="text-[11px] text-amber-400 hover:underline font-medium"
                        >
                          Pilih Semua
                        </button>
                        <button
                          onClick={() => setMvSelectedPaths(new Set())}
                          className="text-[11px] opacity-70 hover:underline"
                        >
                          Reset
                        </button>
                      </div>
                    </div>

                    <input
                      type="text"
                      value={mvFileSearch}
                      onChange={(e) => setMvFileSearch(e.target.value)}
                      placeholder="Cari nama file di repo asal..."
                      className={`w-full px-3 py-1.5 rounded-lg border text-xs font-mono ${
                        isDark ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-300'
                      }`}
                    />

                    <div className="max-h-56 overflow-y-auto divide-y divide-slate-800/40 text-xs font-mono">
                      {Object.values(mvSrcFilesMap)
                        .filter((f) => !mvFileSearch || f.path.toLowerCase().includes(mvFileSearch.toLowerCase()))
                        .map((f) => {
                          const checked = mvSelectedPaths.has(f.path);
                          return (
                            <label
                              key={f.path}
                              className={`py-2 px-2 flex items-center justify-between gap-2 cursor-pointer rounded transition ${
                                checked
                                  ? 'bg-indigo-500/15 text-indigo-300'
                                  : isDark ? 'hover:bg-slate-900' : 'hover:bg-slate-100'
                              }`}
                            >
                              <div className="flex items-center gap-2.5 min-w-0">
                                <input
                                  type="checkbox"
                                  checked={checked}
                                  onChange={() => toggleSelectFile(f.path)}
                                  className="rounded border-slate-600 text-indigo-600"
                                />
                                <span className="truncate font-medium">{f.path}</span>
                              </div>
                              <span className="text-[11px] text-emerald-400 shrink-0">{formatSize(f.size)}</span>
                            </label>
                          );
                        })}
                    </div>
                  </div>
                )}
              </div>

              {/* PANEL KANAN: TUJUAN, RENAME & ANTI-OVERWRITE GUARD */}
              <div className={`rounded-2xl border p-5 shadow-sm space-y-4 ${
                isDark ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'
              }`}>
                <div className="flex items-center justify-between flex-wrap gap-2">
                  <h2 className="font-bold text-sm sm:text-base flex items-center gap-2">
                    <ArrowRightLeft className="w-5 h-5 text-emerald-400" />
                    2. Tujuan (Repo Target), Ganti Nama & Proteksi
                  </h2>
                  {userRepos.length > 0 && (
                    <select
                      onChange={(e) => {
                        if (!e.target.value) return;
                        const found = userRepos.find((r) => r.id === e.target.value);
                        if (found) {
                          setMvDstRepoInput(found.id);
                          setMvDstType(found.type);
                        }
                      }}
                      value=""
                      className={`px-3 py-1 rounded-xl border text-xs ${
                        isDark ? 'bg-slate-950 border-slate-800 text-emerald-400' : 'bg-slate-50 border-slate-300'
                      }`}
                    >
                      <option value="">⚡ Pilih Repo Tujuan Saya...</option>
                      {userRepos.map((r) => (
                        <option key={`dst-${r.type}-${r.id}`} value={r.id}>
                          [{r.type.toUpperCase()}] {r.id}
                        </option>
                      ))}
                    </select>
                  )}
                </div>

                <div className="space-y-3">
                  <div>
                    <label className="block text-xs font-medium mb-1 opacity-80">
                      Repo Tujuan (`username/nama-repo` — otomatis dibuat jika belum ada)
                    </label>
                    <input
                      type="text"
                      list="user-hf-repos"
                      value={mvDstRepoInput}
                      onChange={(e) => setMvDstRepoInput(e.target.value)}
                      placeholder="SEVUNX/aestkr2"
                      className={`w-full px-3.5 py-2 rounded-xl border text-xs font-mono ${
                        isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                      }`}
                    />
                  </div>

                  <div className="grid grid-cols-2 gap-3">
                    <div>
                      <label className="block text-xs font-medium mb-1 opacity-80">Tipe Repo Tujuan</label>
                      <select
                        value={mvDstType}
                        onChange={(e) => setMvDstType(e.target.value as RepoType)}
                        className={`w-full px-3 py-2 rounded-xl border text-xs ${
                          isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                        }`}
                      >
                        <option value="model">model</option>
                        <option value="dataset">dataset</option>
                        <option value="space">space</option>
                      </select>
                    </div>
                    <div>
                      <label className="block text-xs font-medium mb-1 opacity-80">Branch Tujuan</label>
                      <input
                        type="text"
                        value={mvDstBranch}
                        onChange={(e) => setMvDstBranch(e.target.value)}
                        className={`w-full px-3 py-2 rounded-xl border text-xs font-mono ${
                          isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                        }`}
                      />
                    </div>
                  </div>

                  <div>
                    <label className="block text-xs font-medium mb-1 opacity-80">
                      Ganti Nama Baru / Folder Tujuan (`target_new_name_or_path` — Opsional)
                    </label>
                    <input
                      type="text"
                      value={mvDstNewName}
                      onChange={(e) => setMvDstNewName(e.target.value)}
                      placeholder="Kosongkan untuk nama asli, ketik nama baru, atau '/' untuk taruh di root"
                      className={`w-full px-3.5 py-2 rounded-xl border text-xs font-mono ${
                        isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                      }`}
                    />
                  </div>

                  {/* Pengaman Anti-Timpa */}
                  <div>
                    <label className="flex items-center gap-1.5 text-xs font-bold mb-1 text-amber-400">
                      <ShieldCheck className="w-4 h-4" />
                      Pengaman Anti-Timpa (Jika Nama File Sudah Ada di Repo Tujuan)
                    </label>
                    <select
                      value={mvCollisionMode}
                      onChange={(e) => setMvCollisionMode(e.target.value as any)}
                      className={`w-full px-3 py-2 rounded-xl border text-xs font-medium ${
                        isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-300'
                      }`}
                    >
                      <option value="STOP_BATALKAN">🛑 STOP_BATALKAN (Aman - Tolak Jika Nama Sudah Ada)</option>
                      <option value="AUTO_RENAME">🔄 AUTO_RENAME (Otomatis Tambah _1, _2, dst)</option>
                      <option value="OVERWRITE_TIMPA">⚠️ OVERWRITE_TIMPA (Timpa File Lama)</option>
                    </select>
                  </div>

                  <div className="pt-1 space-y-2 text-xs">
                    <label className="flex items-center gap-2.5 cursor-pointer">
                      <input
                        type="checkbox"
                        checked={mvPrivateTarget}
                        onChange={(e) => setMvPrivateTarget(e.target.checked)}
                        className="rounded border-slate-600 text-indigo-600"
                      />
                      <span>Buat sebagai <strong>Private Repo</strong> (jika repo tujuan belum ada)</span>
                    </label>

                    <label className="flex items-center gap-2.5 cursor-pointer text-amber-400 font-medium">
                      <input
                        type="checkbox"
                        checked={mvDeleteSource}
                        onChange={(e) => setMvDeleteSource(e.target.checked)}
                        className="rounded border-slate-600 text-amber-500"
                      />
                      <span>
                        🗑️ <strong>Hapus dari Repo Asal (Move / Cut)</strong> setelah file terverifikasi 100% aman di tujuan
                      </span>
                    </label>
                  </div>
                </div>

                {/* Pratinjau Pemetaan File */}
                {plannedMoveMappings.length > 0 && (
                  <div className={`p-3 rounded-xl border text-[11px] font-mono space-y-1 ${
                    isDark ? 'bg-slate-950/80 border-slate-800' : 'bg-slate-50 border-slate-200'
                  }`}>
                    <div className="font-sans font-bold text-indigo-400 mb-1">
                      📋 Pratinjau Pemetaan ({plannedMoveMappings.length} file):
                    </div>
                    {plannedMoveMappings.slice(0, 6).map((m, i) => (
                      <div key={i} className="flex items-center gap-2 break-all">
                        <span className="opacity-75">{m.src}</span>
                        <ArrowRight className="w-3.5 h-3.5 text-amber-400 shrink-0" />
                        <span className="text-emerald-400 font-bold">{m.dst}</span>
                      </div>
                    ))}
                    {plannedMoveMappings.length > 6 && (
                      <div className="opacity-60">... dan {plannedMoveMappings.length - 6} file lainnya</div>
                    )}
                  </div>
                )}

                <button
                  onClick={handleExecuteMoveOrCopy}
                  disabled={mvStatus.loading}
                  className="w-full py-3 px-5 rounded-xl bg-amber-500 hover:bg-amber-400 disabled:opacity-50 text-slate-950 font-bold text-xs shadow-lg shadow-amber-500/25 flex items-center justify-center gap-2 transition"
                >
                  <ArrowRightLeft className="w-4 h-4" />
                  {mvStatus.loading
                    ? 'Memproses Transfer Server-Side...'
                    : mvDeleteSource
                    ? '✂️ Eksekusi Pindahkan (Move) Sekarang'
                    : '📋 Eksekusi Salin (Copy) Sekarang'}
                </button>

                {/* Log Eksekusi Mover */}
                {(mvStatus.logs.length > 0 || mvStatus.error) && (
                  <div className={`p-3.5 rounded-xl border text-xs font-mono space-y-1.5 ${
                    isDark ? 'bg-slate-950 border-slate-800' : 'bg-slate-50 border-slate-200'
                  }`}>
                    {mvStatus.logs.map((l, i) => (
                      <div key={i}>{l}</div>
                    ))}
                    {mvStatus.error && (
                      <div className="text-red-400 font-bold pt-1">❌ {mvStatus.error}</div>
                    )}
                    {mvStatus.successUrl && (
                      <div className="pt-2 flex flex-wrap gap-3 font-sans font-bold">
                        <a
                          href={mvStatus.successUrl}
                          target="_blank"
                          rel="noreferrer"
                          className="inline-flex items-center gap-1 text-emerald-400 hover:underline"
                        >
                          🔗 Buka Repo Tujuan <ExternalLink className="w-3.5 h-3.5" />
                        </a>
                        {mvStatus.downloadUrl && (
                          <a
                            href={mvStatus.downloadUrl}
                            target="_blank"
                            rel="noreferrer"
                            className="inline-flex items-center gap-1 text-indigo-400 hover:underline"
                          >
                            📥 Link Download Langsung <ExternalLink className="w-3.5 h-3.5" />
                          </a>
                        )}
                      </div>
                    )}
                  </div>
                )}
              </div>
            </div>
          </div>
        )}
      </main>
    </div>
  );
}
